                      </div>

                      <BodyCopyText className="text-gray-600 leading-relaxed">
                        {activeFeature.description}
                      </BodyCopyText>
                    </div>

                </div>
              </motion.div>
            ) : (
              // Optional: Placeholder to prompt user (remove if you want it completely collapsible)
              <motion.div
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                className="text-center  text-gray-400 italic"
              >
                Click on the red pulsing points above to view component details.
              </motion.div>
            )}
          </AnimatePresence>
        </ContainerSection>

      </div>
    </MainPage>
  );
}
